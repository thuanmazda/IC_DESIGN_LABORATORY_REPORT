\section{Mục tiêu thí nghiệm (Objectives)}
\begin{itemize}
    \item Hiểu rõ tất cả các vùng hoạt động của MOSFET thông qua việc phân tích các đặc tuyến $I_D-V_{GS}$ và $I_D-V_{DS}$ của các transistor \texttt{nmos1v\_lvt} và \texttt{pmos1v\_lvt} trong công nghệ \texttt{gpdk045}.
    \item Làm rõ các hiệu ứng bậc hai (Second-order effects): Hiệu ứng thân (Body effect) và Hiệu ứng điều chế chiều dài kênh (Channel length modulation).
    \item Khảo sát đặc tuyến truyền đạt điện áp (VTC) của cổng đảo CMOS.
\end{itemize}

\section{Chuẩn bị thí nghiệm (Preparation for Lab 4)}
Trong phần này, nhóm ôn tập về các vùng hoạt động của MOSFET (Cut-off, Triode, Saturation) với các điều kiện điện áp dòng tương ứng.

\subsection{NMOS Transistor}
\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{|c|c|c|c|}
        \hline
        \textbf{Parameter} & \textbf{Cut-off} & \textbf{Triode} & \textbf{Saturation} \\ \hline
        $I_D$ & $0$ & $\mu_n C_{ox} \frac{W}{L} \left[ (V_{GS}-V_{THN})V_{DS} - \frac{V_{DS}^2}{2} \right]$ & $\frac{1}{2} \mu_n C_{ox} \frac{W}{L} (V_{GS}-V_{THN})^2 (1 + \lambda V_{DS})$ \\ \hline
        $V_{GS}$ & $V_{GS} < V_{THN}$ & $V_{GS} \ge V_{THN}$ & $V_{GS} \ge V_{THN}$ \\ \hline
        $V_{DS}$ & $-$ & $V_{DS} < V_{GS} - V_{THN}$ & $V_{DS} \ge V_{GS} - V_{THN}$ \\ \hline
    \end{tabular}
    \caption{Các điều kiện hoạt động của transistor NMOS}
\end{table}

\subsection{PMOS Transistor}
\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{|c|c|c|c|}
        \hline
        \textbf{Parameter} & \textbf{Cut-off} & \textbf{Triode} & \textbf{Saturation} \\ \hline
        $I_D$ & $0$ & $\mu_p C_{ox} \frac{W}{L} \left[ (V_{SG}-|V_{THP}|)V_{SD} - \frac{V_{SD}^2}{2} \right]$ & $\frac{1}{2} \mu_p C_{ox} \frac{W}{L} (V_{SG}-|V_{THP}|)^2 (1 + \lambda V_{SD})$ \\ \hline
        $V_{SG}$ & $V_{SG} < |V_{THP}|$ & $V_{SG} \ge |V_{THP}|$ & $V_{SG} \ge |V_{THP}|$ \\ \hline
        $V_{SD}$ & $-$ & $V_{SD} < V_{SG} - |V_{THP}|$ & $V_{SD} \ge V_{SG} - |V_{THP}|$ \\ \hline
    \end{tabular}
    \caption{Các điều kiện hoạt động của transistor PMOS}
\end{table}

\textbf{Nhận xét NMOS (Quét $V_{DS}$):}
\begin{itemize}
    \item \textbf{Ước lượng $V_{TH}$:} Từ bảng thông số và đồ thị phóng to, ta thấy $V_{THN} \approx 487 \text{ mV} - 511 \text{ mV}$ tùy thuộc vào mức $V_{DS}$ (có sự sụt giảm nhẹ $V_{TH}$ khi $V_{DS}$ tăng do Drain Induced Barrier Lowering - DIBL).
    \item \textbf{Vùng hoạt động:} Chỉ có 2 vùng hoạt động rõ rệt trên đồ thị này. Khi $V_{GS} < V_{TH}$, linh kiện ở vùng \textbf{Cut-off} ($I_D \approx 0$). Khi $V_{GS} > V_{TH}$, kênh dẫn hình thành, linh kiện đi vào vùng \textbf{Saturation} (dòng $I_D$ tăng mạnh theo hàm mũ/bậc hai).
\end{itemize}

    \item Khi $V_{SB}$ thay đổi, vùng hoạt động vẫn tuân thủ theo nguyên lý cũ (Triode ở $V_{DS}$ nhỏ và Saturation ở $V_{DS}$ lớn), nhưng biên độ dòng bão hòa bị sụt giảm mạnh nếu $V_{SB}$ dương do $V_{OV}$ bị thu hẹp.
\end{itemize}

\subsection{Thảo luận về việc triệt tiêu Body Effect}
\textbf{Câu hỏi định hướng:} \textit{"We can always connect the body terminal to the source terminal of MOSFET to ignore the body effect."}
\begin{itemize}
    \item \textbf{Khẳng định:} Đây là nhận định \textbf{SAI} đối với các quy trình chế tạo CMOS chuẩn.
    \item \textbf{Giải thích:} PMOS nằm trong các giếng N-Well độc lập, do đó ta có thể dễ dàng nối chập cực Body và Source của từng PMOS để ép $V_{BS} = 0$. Tuy nhiên, với NMOS, tất cả các linh kiện đều dùng chung đế P-Substrate (nối đất 0V). Khi có các NMOS mắc nối tiếp (ví dụ NAND gate), NMOS ở trên sẽ có Source $> 0V$ trong khi Body $= 0V$, dẫn đến $V_{SB} > 0$. Body effect ở NMOS trong mạch Stacked là không thể triệt tiêu.
\end{itemize}

\subsection{Đánh giá định lượng và Thảo luận kết quả}
Dựa trên 3 tập đồ thị và 2 bảng thông số trích xuất ở trên, nhóm tiến hành phân tích các đặc tính hình học và vật lý của linh kiện:

\begin{itemize}
    \item \textbf{Xác định Overdrive Voltage ($V_{OV}$ - điện áp quá vận):} Điểm uốn uốn phân tách giữa vùng Triode (tuyến tính) và vùng Saturation (bão hòa) trên các đường cong đặc tuyến chính là điện áp bão hòa $V_{DS,sat} = V_{GS} - V_{THN}$. Đối với đường quét tại $V_{GS} = 600\text{ mV}$, điểm Pinch-off xuất hiện trong khoảng $V_{DS} \approx 110\text{ mV} - 140\text{ mV}$, hoàn toàn khớp với giá trị điện áp ngưỡng $V_{TH}$ được ghi nhận trên hệ thống hiển thị dữ liệu chi tiết.
    \item \textbf{Channel Length Modulation - CLM (Hiệu ứng điều chế chiều dài kênh):} Trong vùng bão hòa ($V_{DS} > V_{OV}$), đặc tuyến dòng điện của các thiết bị có chiều dài ngắn ($L = 45\text{ nm}$) không duy trì trạng thái hằng số nằm ngang lý tưởng mà dốc lên khá rõ do sự mở rộng của vùng nghèo phía cực máng làm rút ngắn chiều dài kênh hiệu dụng.
    \item \textbf{Giải pháp thiết kế giảm độ dốc dòng bão hòa:} Do hệ số điều chế chiều dài kênh $\lambda$ tỉ lệ nghịch với chiều dài hình học kênh dẫn ($\lambda \propto 1/L$), nội trở ngõ ra bão hòa $r_o$ tăng tỉ lệ thuận với độ dài linh kiện ($r_o = \frac{1}{\lambda I_D} \propto L$). Trực quan hóa từ \textit{Graph 3}, đặc tuyến dòng điện của transistor dịch chuyển phẳng dần khi tăng $L$ từ $45\text{ nm}$ lên $450\text{ nm}$. Vì vậy, phương án tối ưu để giảm độ dốc dòng bão hòa trong thiết kế cấu trúc Current Mirror (nguồn dòng gương) là **tăng chiều dài kênh $L$ của linh kiện**.
\end{itemize}

\textbf{Biện luận và Chứng minh lý thuyết:}
\begin{itemize}
    \item \textbf{Vùng hoạt động tại mức điểm $V_{GS} = V_{DS} = 500\text{ mV}$:} 
    Với mức điện áp ngưỡng ước lượng $V_{THN} \approx 380\text{ mV}$, ta tính được điện áp quá vận $V_{OV} = V_{GS} - V_{THN} = 500 - 380 = 120\text{ mV}$. 
    \begin{itemize}
        \item Đối với linh kiện đơn $M_3$: Do thỏa mãn điều kiện $V_{DS} = 500\text{ mV} \ge V_{OV} = 120\text{ mV}$, nên $M_3$ hoạt động ở vùng \textbf{Saturation (Bão hòa)}.
        \item Đối với cặp xếp chồng: Linh kiện phía trên $M_2$ hoạt động ở vùng \textbf{Saturation (Bão hòa)} chịu trách nhiệm ghim dòng bão hòa cho toàn nhánh, trong khi linh kiện phía dưới $M_1$ có điện áp nút trung gian nhỏ nên hoạt động ở vùng \textbf{Triode (Tuyến tính)}.
    \end{itemize}

    \item \textbf{Tính tương đương của hai trường hợp:} 
    Kết quả mô phỏng cho thấy hai kịch bản có đặc tuyến dòng điện trùng khít hoàn toàn lên nhau. Điều này được chứng minh bằng lý thuyết giải tích mạch như sau:
    
    Gọi $V_X$ là điện áp tại nút trung gian giữa $M_1$ và $M_2$. Giả sử hai transistor lý tưởng có cùng hệ số công nghệ $\beta = \mu_n C_{ox} W$.
    
    Dòng điện chạy qua $M_1$ (vùng Tuyến tính):
    $$ I_{D1} = \beta \frac{1}{L} \left[ (V_{GS} - V_{THN})V_X - \frac{V_X^2}{2} \right] $$
    Dòng điện chạy qua $M_2$ (vùng Tuyến tính/Bão hòa tổng quát):
    $$ I_{D2} = \beta \frac{1}{L} \left[ (V_{GS} - V_X - V_{THN})(V_{DS} - V_X) - \frac{(V_{DS} - V_X)^2}{2} \right] $$
    Vì $M_1$ và $M_2$ mắc nối tiếp nên dòng điện qua chúng phải bằng nhau ($I_{D1} = I_{D2} = I_D$). Cộng hai vế phương trình dòng điện của $I_{D1}$ và $I_{D2}$, ta triệt tiêu được các thành phần chứa điện áp nút trung gian $V_X$:
    $$ 2I_D = I_{D1} + I_{D2} = \beta \frac{1}{L} \left[ (V_{GS} - V_{THN})V_{DS} - \frac{V_{DS}^2}{2} \right] $$
    $$ \implies I_D = \mu_n C_{ox} \frac{W}{2L} \left[ (V_{GS} - V_{THN})V_{DS} - \frac{V_{DS}^2}{2} \right] $$
    
    Phương trình cấu trúc dòng điện này hoàn toàn trùng khớp với phương trình dòng của một transistor đơn duy nhất có chiều dài kênh bằng tổng hai chiều dài thành phần $L_{eq} = L_1 + L_2 = 2L$. Với $L = 45\text{ nm}$, chiều dài tương đương chính xác bằng $90\text{ nm}$, tương ứng với thông số hình học của linh kiện $M_3$. Do đó, về mặt bản chất dòng điện, hai test case cho đồ thị đặc tính hoàn toàn giống nhau.
\end{itemize}

\section{Kết luận chung}
Thông qua chuỗi bài thực hành chuyên sâu về khảo sát và định lượng đặc tính linh kiện MOSFET trên tiến trình công nghệ vi mạch bBulk \texttt{gpdk045} ($45\text{ nm}$), nhóm đã hoàn thành toàn bộ các yêu cầu kỹ thuật và rút ra các kết luận tổng quan mang tính cốt lõi sau:

\begin{enumerate}
    \item \textbf{Sự tương thích giữa thực nghiệm và lý thuyết hoạt động:} Các đồ thị đặc tuyến truyền đạt $I_D - V_{GS}$ và đặc tuyến ngõ ra $I_D - V_{DS}$ thu được từ Cadence Virtuoso phản ánh hoàn hảo cấu trúc toán học của các vùng hoạt động: Cut-off (nghèo), Triode (tuyến tính), và Saturation (bão hòa) cho cả hai cấu trúc linh kiện nghịch (\texttt{nmos1v\_lvt}) và thuận (\texttt{pmos1v\_lvt}).
    \item \textbf{Vai trò điều khiển của kích thước hình học:} Thực nghiệm quét tham số kích thước mặt bằng đã chứng minh rõ nét bản chất vật lý của linh kiện:
    \begin{itemize}
        \item Chiều rộng kênh $W$ tỉ lệ thuận với độ dốc (Transconductance - $g_m$), điều khiển biên độ chịu dòng tối đa của thiết bị mà không làm dịch chuyển điện áp ngưỡng $V_{TH}$.
        \item Chiều dài kênh $L$ là tham số cốt lõi chi phối nội trở ngõ ra bão hòa $r_o$. Khi chiều dài hình học $L$ càng ngắn, hiệu ứng bậc hai CLM xảy ra càng khốc liệt khiến đồ thị dòng bão hòa bị dốc lên đáng kể. Do đó, tăng chiều dài $L$ là giải pháp thiết kế bắt buộc khi cấu trúc các nguồn dòng gương (Current Mirror) yêu cầu độ ổn định cao.
    \end{itemize}
    \item \textbf{Làm rõ bản chất các hiệu ứng bậc hai (Second-order effects):} 
    \begin{itemize}
        \item \textit{Body Effect (Hiệu ứng thân):} Việc phân cực ngược mối nối Source - Body ($V_{SB} > 0$) làm mở rộng vùng nghèo ký sinh, gián tiếp làm tăng điện áp ngưỡng $V_{TH}$ và làm suy giảm điện áp quá vận $V_{OV}$ của linh kiện. Việc phản biện thành công giả thuyết luôn loại bỏ được Body effect giúp nhóm thấu hiểu sâu sắc các ràng buộc vật lý về chia sẻ đế P-substrate chung trong các cổng logic tĩnh Standard CMOS.
        \item \textit{Channel Length Modulation (CLM - Hiệu ứng điều chế chiều dài kênh):} Đã định lượng được sự không lý tưởng của nguồn dòng thông qua độ dốc của đồ thị $I_D$. Việc uốn cong điện áp bão hòa thực tế tại điểm uốn $V_{DS,sat}$ giúp nhóm xác định chính xác ranh giới phân cực tĩnh cho linh kiện.
    \end{itemize}
    \item \textbf{Ý nghĩa của cấu trúc Stacked và Multiplier:} Kết quả mô phỏng so sánh hai mô hình phân nhánh nối tiếp và song song đã kiểm chứng trực quan các định luật mạch kinh điển (KCL, điện trở tương đương) trong môi trường vi mạch thực tế. Sự xuất hiện của Body effect ký sinh tại node chung của cấu trúc xếp chồng chính là bài học định lượng đắt giá về việc tính toán sụt giảm dòng điện mà lý thuyết sơ cấp thường bỏ qua.
\end{enumerate}

Những kết quả trực quan và số liệu tra cứu chi tiết thu được từ các cửa sổ hiển thị dữ liệu trong bài thí nghiệm này chính là nền tảng kỹ thuật vững chắc, giúp nhóm làm chủ phương pháp ước lượng, tính toán và tối ưu hóa kích thước hình học linh kiện cho các thiết kế mạch tích hợp tĩnh và tương tự phức tạp hơn ở các chương tiếp theo.
